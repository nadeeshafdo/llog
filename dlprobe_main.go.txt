package main

import (
	"bufio"
	"fmt"
	"os"
	"time"

	"golang.org/x/sys/unix"
	"golang.org/x/term"
)

func main() {
	master, _ := os.OpenFile("/dev/ptmx", os.O_RDWR, 0)
	defer master.Close()
	var zero int32
	_ = zero
	unix.IoctlSetPointerInt(int(master.Fd()), unix.TIOCSPTLCK, 0)
	ptn, _ := unix.IoctlGetInt(int(master.Fd()), unix.TIOCGPTN)
	slave, _ := os.OpenFile(fmt.Sprintf("/dev/pts/%d", ptn), os.O_RDWR, 0)
	defer slave.Close()
	st, _ := term.MakeRaw(int(slave.Fd()))
	defer term.Restore(int(slave.Fd()), st)

	master.Write([]byte{0x1b})
	r := bufio.NewReader(slave)
	b, _, err := r.ReadRune()
	fmt.Printf("first byte: %q err=%v\n", string(b), err)

	t0 := time.Now()
 derr := slave.SetReadDeadline(time.Now().Add(50 * time.Millisecond))
	fmt.Printf("setDeadline err=%v elapsed=%v\n", derr, time.Since(t0))
	slave.SetReadDeadline(time.Time{})

	t1 := time.Now()
	derr2 := slave.SetReadDeadline(time.Now().Add(100 * time.Millisecond))
	fmt.Printf("setDeadline2 err=%v\n", derr2)
	_, _, rerr := r.ReadRune()
	fmt.Printf("second read err=%v elapsed=%v\n", rerr, time.Since(t1))
}
